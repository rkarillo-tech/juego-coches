Conectar Google Drive en 2 minutos (100% Gratis)

  1. Entra en script.google.com con tu cuenta de Gmail y pulsa "Nuevo proyecto".
  2. Borra lo que haya y pega este código de 18 líneas (crea y actualiza el archivo garage_legends_cloud.json en tu propio Google Drive):
function doPost(e) {
  var fileName = "garage_legends_cloud.json";
  var files = DriveApp.getFilesByName(fileName);
  if (files.hasNext()) {
    files.next().setContent(e.postData.contents);
  } else {
    DriveApp.createFile(fileName, e.postData.contents);
  }
  return ContentService.createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}
function doGet() {
  var files = DriveApp.getFilesByName("garage_legends_cloud.json");
  var data = files.hasNext() ? files.next().getBlob().getDataAsString() : "{}";
  return ContentService.createTextOutput(data)
    .setMimeType(ContentService.MimeType.JSON);
}
